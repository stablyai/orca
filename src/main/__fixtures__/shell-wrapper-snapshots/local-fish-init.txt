function __orca_shell_ready_marker --on-event fish_prompt
  builtin printf "\033]777;orca-shell-ready\007"
  functions -e __orca_shell_ready_marker
end
# Why captured: an unquoted (type -t codex) expands to zero words when codex is
# absent, leaving "test = file" — fish then errors instead of failing closed.
# Quoting in place is not the fix; fish never substitutes inside double quotes.
set -l __orca_codex_type (type -t codex 2>/dev/null)
if test "$__orca_codex_type" = file
  function codex
    if test -x "$ORCA_CODEX_LAUNCH_PREFLIGHT"
      command "$ORCA_CODEX_LAUNCH_PREFLIGHT" agent hooks prepare-codex >/dev/null 2>&1; or true
    end
    if test "$ORCA_CODEX_ISOLATE" != 0; and not string match -qr -- '^(agents|queue|--remote|--remote=.*)$' $argv
      test -n "$ORCA_TERMINAL_HANDLE"; and not string match -q -- "*'*" "$ORCA_TERMINAL_HANDLE"; and set argv -c "shell_environment_policy.set.ORCA_TERMINAL_HANDLE='$ORCA_TERMINAL_HANDLE'" $argv
      test -n "$ORCA_PANE_KEY"; and not string match -q -- "*'*" "$ORCA_PANE_KEY"; and set argv -c "shell_environment_policy.set.ORCA_PANE_KEY='$ORCA_PANE_KEY'" $argv
      test -n "$ORCA_WORKTREE_ID"; and not string match -q -- "*'*" "$ORCA_WORKTREE_ID"; and set argv -c "shell_environment_policy.set.ORCA_WORKTREE_ID='$ORCA_WORKTREE_ID'" $argv
      test -n "$ORCA_WORKSPACE_ID"; and not string match -q -- "*'*" "$ORCA_WORKSPACE_ID"; and set argv -c "shell_environment_policy.set.ORCA_WORKSPACE_ID='$ORCA_WORKSPACE_ID'" $argv
      test -n "$ORCA_USER_DATA_PATH"; and not string match -q -- "*'*" "$ORCA_USER_DATA_PATH"; and set argv -c "shell_environment_policy.set.ORCA_USER_DATA_PATH='$ORCA_USER_DATA_PATH'" $argv
    end
    if test "$ORCA_CODEX_ISOLATE" != 0; and not string match -qr -- '^(agents|queue|--remote|--no-daemon|--remote=.*)$' $argv; and command codex --help 2>/dev/null </dev/null | string match -q -- '*--no-daemon*'
      set argv --no-daemon $argv
    end
    if test "$ORCA_REMOTE_VIEWER_CLIPBOARD" = 1; and not set -q SSH_CONNECTION; and not set -q SSH_TTY
      set -lx SSH_CONNECTION '127.0.0.1 0 127.0.0.1 0'
      command codex $argv
    else
      command codex $argv
    end
  end
end
set -e __orca_codex_type
set -l __orca_claude_type (type -t claude 2>/dev/null)
if test -n "$ORCA_CLAUDE_PROFILE_POINTER"; and test "$__orca_claude_type" = file
  function claude
    # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
    set -l pointer (string replace -r '^~/' "$HOME/" -- "$ORCA_CLAUDE_PROFILE_POINTER")
    set -l profile (cat "$pointer" 2>/dev/null)
    set -l own (string trim -r -c / -- "$ORCA_CLAUDE_USER_CONFIG_DIR")
    set -l current (string trim -r -c / -- "$CLAUDE_CONFIG_DIR")
    if test -n "$CLAUDE_CONFIG_DIR"; and test "$CLAUDE_CONFIG_DIR" != "$ORCA_CLAUDE_INJECTED_CONFIG_DIR"; and begin; test -z "$own"; or test "$current" != "$own"; end
      command claude $argv
      return $status
    end
    if test -z "$profile"; and test -n "$CLAUDE_CONFIG_DIR"; and test -n "$ORCA_CLAUDE_USER_CONFIG_DIR"
      env -u ORCA_CLAUDE_INJECTED_CONFIG_DIR CLAUDE_CONFIG_DIR="$ORCA_CLAUDE_USER_CONFIG_DIR" claude $argv
      return $status
    end
    if test -z "$profile"
      env -u CLAUDE_CONFIG_DIR -u ORCA_CLAUDE_INJECTED_CONFIG_DIR claude $argv
      return $status
    end
    if not string match -qr '^(/|[A-Za-z]:)' -- "$profile"
      return 1
    end
    test -d "$profile"; or mkdir -p -m 700 -- "$profile" 2>/dev/null
    env CLAUDE_CONFIG_DIR="$profile" ORCA_CLAUDE_INJECTED_CONFIG_DIR="$profile" claude $argv
  end
end
set -e __orca_claude_type
